; data
PARTS_DEFINE:
	; ATTR:
	;	Ypos: 4bit
	;		Ypos == f: end
	;	repeatX: 1bit
	;	repeatY: 1bit
	;	1block: 1bit
	;	nextLine: 1bit
	; repeatフラグは二つセット可能，その間をリピートする
	.byte 'b', (ATTR $4, 0, 0, 0, 1)
	.byte 'b', (ATTR $f, 0, 0, 0, 0)

; addr-table (split)
PARTS_LO:
	.byte <PARTS_0
		...
	.byte <PARTS_21
PARTS_SP12:
	.byte <PARTS_22
		...

PARTS_HI:
	.byte >PARTS_0
		...
	.byte >PARTS_21
	.byte >PARTS_22

; table for sp obj
PARTS_SP_INDEX_CALC_TABLE:
	.byte PARTS_SP12 - PARTS_LO


; prg
func setAddr(index) {
	addr_hi = PARTS_HI[index]
	addr_lo = PARTS_LO[index]
	; (addr_hi), y = *(PARTS_DEFINE+y)
}
func spObjSet(type, y) {
	; PARTS_SP_INDEX_CALC_TABLEのインデックスは最大4種類
	; → 条件分岐で，定数として読み込んでも良い
	index = type + PARTS_SP_INDEX_CALC_TABLE[y-12]
	setAddr(index, y)
	posY = PARTS_YPOS[index]
}

背景オブジェクト
・雲
・草（タイルは雲を流用できる）
・低い木（4回出現，タイルは8枚で構成）
・高い木（3回出現，タイルは12枚で構成）
・大きい山（一回しか登場しない）
・小さい山（一回しか登場しない）
・柵（5-6回登場）

山はちょっとだけ無駄になるが，すべてオブジェクト化
背景パターンのデータには，xy座標とオブジェクト番号
座標とオブジェクト番号を2Byteで管理
	.byte %ff0xxxxx
	.byte %yyy00iii

	; ff: フラグ
	;	0: 通常
	;	2: 次の画面から描画
	;	3: マップデータの末尾，最初へ戻る
	; xxxxx: X座標（8*8タイル単位）
	; x座標をdxxxxxとみることもできる

	; iii: オブジェクト番号
	;	0: 雲/草1つ
	;	1: 雲/草2つ連結
	;	2: 雲/草3つ連結
	;		yyy: y座標のパターン(0-4, 5は草に使用)
	;	3: 低い木
	;	4: 高い木
	;	5: 大きい山
	;	6: 小さい山
	;	7: 柵
	;		3-7はyyy = 5


位置が固定なので，パレットデータはオブジェクトと関連付けずに持っておく
.byte dxxx0yyy
.byte aabbccdd
; d: 画面番号->次の画面フラグへ変更
; xxx: X座標（32*32のパレット単位）
; yyy: Y座標
; aabbccdd: パレットのデータ（オブジェクトのない領域も含め4ブロック分のパレット）


オブジェクトは
	.byte hhhiiiii

	; hhh: オブジェクトy（8*8タイル単位）
	;	hhh = 0のとき，オブジェクトの一番上側，数が大きくなるほど下へ
	;	hhh = 7で次の画面へ
	; p: x++
	; iiiii: タイル番号
	;	%000iiiii + $afが実際のタイル番号
	;	af-c7がBGのタイル番号に割り当て，後から拡張されることはないはず

（すべてのオブジェクトはx座標が被らない）
